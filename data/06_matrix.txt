示例：matrix = [[1,1,1],[1,0,1],[1,1,1]] → 输出 [[1,0,1],[0,0,0],[1,0,1]]。
@@SOLUTION: 方法一：首行首列作标记（O(1) 空间，推荐）
@@IDEA
O(1) 空间标准解：用第一行和第一列本身作标记数组。注意先单独记录首行首列自身是否含 0，最后再处理它们，避免标记被提前覆盖。O(mn) 时间。
@@JAVA
class Solution {
    public void setZeroes(int[][] matrix) {
        int m = matrix.length, n = matrix[0].length;
        boolean row0 = false, col0 = false;
        for (int j = 0; j < n; j++) if (matrix[0][j] == 0) row0 = true;
        for (int i = 0; i < m; i++) if (matrix[i][0] == 0) col0 = true;
        for (int i = 1; i < m; i++) {
            for (int j = 1; j < n; j++) {
                if (matrix[i][j] == 0) { // 标记到首行首列
                    matrix[i][0] = 0;
                    matrix[0][j] = 0;
                }
            }
        }
        for (int i = 1; i < m; i++) {
            for (int j = 1; j < n; j++) {
                if (matrix[i][0] == 0 || matrix[0][j] == 0) matrix[i][j] = 0;
            }
        }
        if (row0) for (int j = 0; j < n; j++) matrix[0][j] = 0;
        if (col0) for (int i = 0; i < m; i++) matrix[i][0] = 0;
    }
}

@@IDEA

class Solution {
    public void setZeroes(int[][] matrix) {
        Set<Integer> rows = new HashSet<>(), cols = new HashSet<>();
        int m = matrix.length, n = matrix[0].length;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (matrix[i][j] == 0) {
                    rows.add(i); // 记录含 0 的行列
                    cols.add(j);
                }
            }
        }
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (rows.contains(i) || cols.contains(j)) {
                    matrix[i][j] = 0;
                }
            }
        }
    }
}

    public void setZeroes(int[][] matrix) {

给你一个 m x n 矩阵 matrix，请按顺时针螺旋顺序返回矩阵中的所有元素。
示例：matrix = [[1,2,3],[4,5,6],[7,8,9]] → 输出 [1,2,3,6,9,8,7,4,5]。
@@SOLUTION: 方法一：四边界逐圈收缩（推荐）
@@IDEA
维护 top/bottom/left/right 四个边界，按「右→下→左→上」走一圈收缩一次；每走完一条边立即检查边界是否交叉，防止剩单行/单列时重复输出。O(mn)。
@@JAVA
class Solution {
    public List<Integer> spiralOrder(int[][] matrix) {
        List<Integer> res = new ArrayList<>();
        int top = 0, bottom = matrix.length - 1;
        int left = 0, right = matrix[0].length - 1;
        while (top <= bottom && left <= right) {
            for (int j = left; j <= right; j++) res.add(matrix[top][j]);
            top++;
            for (int i = top; i <= bottom; i++) res.add(matrix[i][right]);
            right--;
            if (top <= bottom) { // 剩单行时避免重复
                for (int j = right; j >= left; j--) res.add(matrix[bottom][j]);
                bottom--;
            }
            if (left <= right) { // 剩单列时避免重复
                for (int i = bottom; i >= top; i--) res.add(matrix[i][left]);
                left++;
            }
        }
        return res;
    }
}

@@SOLUTION: 方法二：方向模拟 + 访问标记
@@IDEA
方向向量模拟：按右下左上顺序走，越界或踩到已访问格子就顺时针转向；用 visited 矩阵防回头。O(mn) 时间与空间，逻辑机械不易错。
@@JAVA
class Solution {
    public List<Integer> spiralOrder(int[][] matrix) {
        int m = matrix.length, n = matrix[0].length;
        boolean[][] visited = new boolean[m][n];
        int[][] dirs = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}}; // 右下左上
        List<Integer> res = new ArrayList<>();
        int i = 0, j = 0, d = 0;
        for (int step = 0; step < m * n; step++) {
            res.add(matrix[i][j]);
            visited[i][j] = true;
            int ni = i + dirs[d][0], nj = j + dirs[d][1];
            if (ni < 0 || ni >= m || nj < 0 || nj >= n || visited[ni][nj]) {
                d = (d + 1) % 4;           // 顺时针转向
                ni = i + dirs[d][0];
                nj = j + dirs[d][1];
            }
            i = ni; j = nj;
        }
        return res;
    }
}

@@SOLUTION: 方法三：递归剥环（Python 另附 zip 极简版）
@@IDEA
递归剥环：每层取顶行、右列、底行反向、左列反向，再对内圈子矩阵递归。Python 有极简写法：剥离首行后把剩余矩阵逆时针旋转 90°（zip）重复即可。
@@JAVA
class Solution {
    public List<Integer> spiralOrder(int[][] matrix) {
        List<Integer> res = new ArrayList<>();
        peel(matrix, 0, matrix.length - 1, 0, matrix[0].length - 1, res);
        return res;
    }
    // 剥一层环，再对内圈递归
    private void peel(int[][] m, int rLo, int rHi, int cLo, int cHi, List<Integer> res) {
        if (rLo > rHi || cLo > cHi) return;
        for (int j = cLo; j <= cHi; j++) res.add(m[rLo][j]);   // 顶行
        for (int i = rLo + 1; i <= rHi; i++) res.add(m[i][cHi]); // 右列
        if (rLo < rHi) {
            for (int j = cHi - 1; j >= cLo; j--) res.add(m[rHi][j]); // 底行反向
        }
        if (cLo < cHi) {
            for (int i = rHi - 1; i > rLo; i--) res.add(m[i][cLo]); // 左列反向
        }
        peel(m, rLo + 1, rHi - 1, cLo + 1, cHi - 1, res);      // 内圈递归
    }
}

@@SOLUTION: 方法一：转置 + 行翻转（推荐）
@@IDEA
顺时针旋转 90° = 先沿主对角线转置，再把每行左右翻转。两步都是原地操作，分开记忆不易错。O(n²)/O(1)。

class Solution {
    public void rotate(int[][] matrix) {
        int n = matrix.length;
        for (int i = 0; i < n; i++) {        // 转置
            for (int j = i + 1; j < n; j++) {
                int t = matrix[i][j]; matrix[i][j] = matrix[j][i]; matrix[j][i] = t;
            }
        }
        for (int[] row : matrix) {           // 每行翻转
            for (int l = 0, r = n - 1; l < r; l++, r--) {
                int t = row[l]; row[l] = row[r]; row[r] = t;
            }
        }
    }
}

class Solution {
    public void rotate(int[][] matrix) {
        int n = matrix.length;
        // 副对角线翻转：swap(matrix[i][j], matrix[n-1-j][n-1-i])
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n - 1 - i; j++) {
                int t = matrix[i][j];
                matrix[i][j] = matrix[n - 1 - j][n - 1 - i];
                matrix[n - 1 - j][n - 1 - i] = t;
            }
        }
        for (int i = 0; i < n / 2; i++) { // 上下翻转
            for (int j = 0; j < n; j++) {
                int t = matrix[i][j];
                matrix[i][j] = matrix[n - 1 - i][j];
                matrix[n - 1 - i][j] = t;
            }
        }
    }
}

@@IDEA
从右上角出发的阶梯查找：当前值偏大说明该列都太大，左移一列；偏小说明该行都太小，下移一行。每步排除一行或一列。O(m+n)，本题最优思路。

class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int m = matrix.length, n = matrix[0].length;
        int i = 0, j = n - 1;   // 从右上角出发
        while (i < m && j >= 0) {
            int v = matrix[i][j];
            if (v == target) return true;
            if (v > target) j--; // 偏大 → 左移
            else i++;            // 偏小 → 下移
        }
        return false;
    }
}

            int lo = 0, hi = row.length - 1;
            while (lo <= hi) {            // 行内二分
                int mid = (lo + hi) >>> 1;
                if (row[mid] == target) return true;
                if (row[mid] < target) lo = mid + 1;
                else hi = mid - 1;
            }
        }
        return false;
    }

@@IDEA
与解法一对称：从左下角出发，偏大上移、偏小右移。同阶梯排除法，O(m+n)，展示矩阵搜索「选好起点」的重要性。

class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int m = matrix.length, n = matrix[0].length;
        int i = m - 1, j = 0;   // 从左下角出发
        while (i >= 0 && j < n) {
            int v = matrix[i][j];
            if (v == target) return true;
            if (v > target) i--; // 偏大 → 上移
            else j++;            // 偏小 → 右移
        }
        return false;
    }
}

class Solution:
    def searchMatrix(self, matrix: List[List[int]], target: int) -> bool:
        m, n = len(matrix), len(matrix[0])
        i, j = m - 1, 0  # 从左下角出发
        while i >= 0 and j < n:

            if v == target:
                return True
            if v > target:
                i -= 1  # 偏大 → 上移
            else:
                j += 1  # 偏小 → 右移
