@@SOLUTION: 方法一：回溯 + used 数组（推荐）
@@IDEA
回溯 + used 数组：每层从 0 开始挑一个未用过的数放入路径，递归后撤销（弹出+复位）；路径满 n 时收集快照。O(n·n!)。
@@JAVA
class Solution {
    public List<List<Integer>> permute(int[] nums) {
        List<List<Integer>> res = new ArrayList<>();
        backtrack(nums, new boolean[nums.length], new ArrayDeque<>(), res);
        return res;
    }
    private void backtrack(int[] nums, boolean[] used, Deque<Integer> path,
                           List<List<Integer>> res) {
        if (path.size() == nums.length) {
            res.add(new ArrayList<>(path)); // 固定快照
            return;
        }
        for (int i = 0; i < nums.length; i++) {
            if (used[i]) continue;
            used[i] = true;
            path.addLast(nums[i]);  // 做选择
            backtrack(nums, used, path, res);
            path.removeLast();      // 撤销选择
            used[i] = false;
        }
    }
}

                path.append(x)    # 做选择

        return res
@@SOLUTION: 方法二：交换法（原地回溯）
@@IDEA
交换法：对位置 pos，依次把每个候选元素换到 pos，递归 pos+1，再换回撤销；无需 used 数组、原地完成。O(n·n!)。
@@JAVA
class Solution {
    public List<List<Integer>> permute(int[] nums) {
        List<List<Integer>> res = new ArrayList<>();
        backtrack(nums, 0, res);
        return res;
    }
    private void backtrack(int[] nums, int pos, List<List<Integer>> res) {
        if (pos == nums.length) {
            List<Integer> one = new ArrayList<>();
            for (int x : nums) one.add(x);
            res.add(one);
            return;
        }
        for (int i = pos; i < nums.length; i++) {
            swap(nums, pos, i);     // 候选换到 pos
            backtrack(nums, pos + 1, res);
            swap(nums, pos, i);     // 换回撤销
        }
    }
    private void swap(int[] a, int i, int j) {
        int t = a[i]; a[i] = a[j]; a[j] = t;
    }
}

class Solution {
    public List<List<Integer>> permute(int[] nums) {
        List<List<Integer>> res = new ArrayList<>();
        res.add(new ArrayList<>()); // 空排列
        for (int x : nums) {
            List<List<Integer>> next = new ArrayList<>();
            for (List<Integer> p : res) {
                for (int i = 0; i <= p.size(); i++) { // 插入到每个空位
                    List<Integer> one = new ArrayList<>(p);
                    one.add(i, x);
                    next.add(one);
                }
            }
            res = next;
        }
        return res;
    }
}

@@IDEA

class Solution {
    public List<List<Integer>> subsets(int[] nums) {
        int n = nums.length;
        List<List<Integer>> res = new ArrayList<>();
        for (int mask = 0; mask < (1 << n); mask++) { // 枚举所有掩码
            List<Integer> subset = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) != 0) subset.add(nums[i]); // 第 i 位选
            }
            res.add(subset);
        }
        return res;
    }
}

@@IDEA

class Solution {
    private static final String[] MAP = {
        "", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"
    };

    public List<String> letterCombinations(String digits) {
        List<String> res = new ArrayList<>();
        if (digits.isEmpty()) return res;
        backtrack(digits, 0, new StringBuilder(), res);
        return res;
    }
    private void backtrack(String digits, int pos, StringBuilder path, List<String> res) {
        if (pos == digits.length()) {
            res.add(path.toString());
            return;
        }
        for (char c : MAP[digits.charAt(pos) - '0'].toCharArray()) {
            path.append(c);                        // 做选择
            backtrack(digits, pos + 1, path, res);
            path.deleteCharAt(path.length() - 1);  // 撤销选择
        }
    }
}

@@IDEA

class Solution {
    public List<String> letterCombinations(String digits) {
        if (digits.isEmpty()) return new ArrayList<>();
        String[] map = {"", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};
        Deque<String> queue = new ArrayDeque<>();
        queue.offer("");
        for (char d : digits.toCharArray()) {
            int size = queue.size();
            for (int i = 0; i < size; i++) {
                String base = queue.poll();
                for (char c : map[d - '0'].toCharArray()) {
                    queue.offer(base + c); // 扩展一位
                }
            }
        }
        return new ArrayList<>(queue);
    }
}

class Solution {
    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        Arrays.sort(candidates);
        List<List<Integer>> res = new ArrayList<>();
        backtrack(candidates, target, 0, new ArrayDeque<>(), res);
        return res;
    }
    private void backtrack(int[] cand, int remain, int start, Deque<Integer> path,
                           List<List<Integer>> res) {
        if (remain == 0) {
            res.add(new ArrayList<>(path));
            return;
        }
        for (int i = start; i < cand.length; i++) {
            if (cand[i] > remain) break; // 剪枝（已排序）
            path.addLast(cand[i]);
            backtrack(cand, remain - cand[i], i, path, res); // i 不 +1：可复用
            path.removeLast();
        }
    }
}

@@IDEA

class Solution {
    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        List<List<Integer>> res = new ArrayList<>();
        dfs(candidates, 0, target, new ArrayDeque<>(), res);
        return res;
    }
    private void dfs(int[] cand, int i, int remain, Deque<Integer> path,
                     List<List<Integer>> res) {
        if (remain == 0) {
            res.add(new ArrayList<>(path));
            return;
        }
        if (i >= cand.length || remain < 0) return;
        path.addLast(cand[i]);                       // 选
        dfs(cand, i, remain - cand[i], path, res);   // 可继续选
        path.removeLast();                           // 不选
        dfs(cand, i + 1, remain, path, res);
    }

@@SOLUTION: 方法三：动态规划构造方案（DP 求组合 + 回溯还原）
@@IDEA
记忆化搜索：dfs(i, remain) 缓存「用 candidates[i..] 凑出 remain 的全部组合」，避免重复子问题重复展开。最坏仍指数，但去重后更快。
@@JAVA
class Solution {
    private final Map<String, List<List<Integer>>> memo = new HashMap<>();

    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        return dfs(candidates, 0, target);
    }
    // 记忆化：返回用 cand[i..] 凑出 remain 的所有组合
    private List<List<Integer>> dfs(int[] cand, int i, int remain) {
        if (remain == 0) {
            List<List<Integer>> base = new ArrayList<>();
            base.add(new ArrayList<>()); // 空组合
            return base;
        }
        if (i >= cand.length || remain < 0) return new ArrayList<>();
        String key = i + "," + remain;
        if (memo.containsKey(key)) return memo.get(key);
        List<List<Integer>> res = new ArrayList<>();
        // 选 cand[i]（可重复）
        for (List<Integer> sub : dfs(cand, i, remain - cand[i])) {
            List<Integer> one = new ArrayList<>();
            one.add(cand[i]);
            one.addAll(sub);
            res.add(one);
        }
        // 跳过 cand[i]
        res.addAll(dfs(cand, i + 1, remain));
        memo.put(key, res);
        return res;
    }
}

class Solution:
    def combinationSum(self, candidates: List[int], target: int) -> List[List[int]]:
        from functools import lru_cache

        @lru_cache(maxsize=None)
        def dfs(i, remain):
            """返回用 candidates[i..] 凑出 remain 的所有组合（元组套元组）"""
            if remain == 0:
                return ((),)
            if i >= len(candidates) or remain < 0:
                return ()
            res = []
            # 选 candidates[i]（可重复）
            for sub in dfs(i, remain - candidates[i]):
                res.append((candidates[i],) + sub)
            # 跳过 candidates[i]
            res.extend(dfs(i + 1, remain))
            return tuple(res)

        return [list(t) for t in dfs(0, target)]

class Solution {
    public List<String> generateParenthesis(int n) {
        Set<String> res = new HashSet<>();
        permute(new char[2 * n], 0, n, n, res);
        return new ArrayList<>(res);
    }
    private void permute(char[] cs, int pos, int open, int close, Set<String> res) {
        if (pos == cs.length) {
            if (valid(cs)) res.add(new String(cs)); // 校验有效性
            return;
        }
        if (open > 0) { cs[pos] = '('; permute(cs, pos + 1, open - 1, close, res); }
        if (close > 0) { cs[pos] = ')'; permute(cs, pos + 1, open, close - 1, res); }
    }
    private boolean valid(char[] cs) {
        int bal = 0;
        for (char c : cs) {
            bal += (c == '(') ? 1 : -1;
            if (bal < 0) return false; // 前缀不能为负
        }
        return bal == 0;
    }

class Solution {
    public boolean exist(char[][] board, String word) {
        int m = board.length, n = board[0].length;
        for (int i = 0; i < m; i++) {

        if (i < 0 || i >= board.length || j < 0 || j >= board[0].length

                || dfs(board, word, i, j + 1, k + 1)

class Solution {
    public boolean exist(char[][] board, String word) {
        int m = board.length, n = board[0].length;
        int[] cnt = new int[128];
        for (char[] row : board) {
            for (char c : row) cnt[c]++;      // 网格字符计数
        }
        int[] need = new int[128];
        for (char c : word.toCharArray()) need[c]++;
        for (int c = 0; c < 128; c++) {
            if (need[c] > cnt[c]) return false; // 字母不够，剪枝
        }
        // 首字母比尾字母出现得多 → 反转词搜索（起点更少）
        if (cnt[word.charAt(0)] > cnt[word.charAt(word.length() - 1)]) {
            word = new StringBuilder(word).reverse().toString();
        }
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (dfs(board, word, i, j, 0)) return true;
            }
        }
        return false;
    }
    private boolean dfs(char[][] board, String word, int i, int j, int k) {
        if (board[i][j] != word.charAt(k)) return false;
        if (k == word.length() - 1) return true;
        board[i][j] = '#';
        boolean found = (i + 1 < board.length && dfs(board, word, i + 1, j, k + 1))
                || (i > 0 && dfs(board, word, i - 1, j, k + 1))
                || (j + 1 < board[0].length && dfs(board, word, i, j + 1, k + 1))
                || (j > 0 && dfs(board, word, i, j - 1, k + 1));
        board[i][j] = word.charAt(k);
        return found;

class Solution {
    public List<List<String>> partition(String s) {
        int n = s.length();
        boolean[][] pal = new boolean[n][n]; // pal[i][j]: s[i..j] 回文?
        for (int j = 0; j < n; j++) {
            for (int i = 0; i <= j; i++) {
                if (s.charAt(i) == s.charAt(j) && (j - i < 2 || pal[i + 1][j - 1])) {
                    pal[i][j] = true; // 区间 DP
                }
            }
        }
        List<List<String>> res = new ArrayList<>();
        backtrack(s, 0, pal, new ArrayDeque<>(), res);
        return res;
    }
    private void backtrack(String s, int start, boolean[][] pal, Deque<String> path,
                           List<List<String>> res) {
        if (start == s.length()) {
            res.add(new ArrayList<>(path));
            return;
        }
        for (int end = start; end < s.length(); end++) {
            if (!pal[start][end]) continue; // 只切回文段
            path.addLast(s.substring(start, end + 1));
            backtrack(s, end + 1, pal, path, res);
            path.removeLast();
        }
    }
}

逐行回溯 + 三组布尔标记：逐行放置天然不同行；col[c]、主对角线 row+c、副对角线 row-c+n 三组标记防冲突，回溯尝试每列。O(n!) 量级。
@@JAVA
class Solution {
    public List<List<String>> solveNQueens(int n) {
        List<List<String>> res = new ArrayList<>();
        char[][] board = new char[n][n];
        for (char[] row : board) java.util.Arrays.fill(row, '.');
        backtrack(board, 0, new boolean[n], new boolean[2 * n], new boolean[2 * n], res);
        return res;
    }
    private void backtrack(char[][] board, int row, boolean[] col,
                           boolean[] diag1, boolean[] diag2, List<List<String>> res) {
        int n = board.length;
        if (row == n) {
            List<String> solution = new ArrayList<>();
            for (char[] r : board) solution.add(new String(r));
            res.add(solution);
            return;
        }
        for (int c = 0; c < n; c++) {
            if (col[c] || diag1[row + c] || diag2[row - c + n]) continue; // 冲突
            board[row][c] = 'Q';
            col[c] = diag1[row + c] = diag2[row - c + n] = true;
            backtrack(board, row + 1, col, diag1, diag2, res);
            board[row][c] = '.';
            col[c] = diag1[row + c] = diag2[row - c + n] = false;
        }
    }
}

class Solution:
    def solveNQueens(self, n: int) -> List[List[str]]:
        res = []
        board = [['.'] * n for _ in range(n)]
        cols = [False] * n
        diag1 = [False] * (2 * n)  # row + c
        diag2 = [False] * (2 * n)  # row - c + n

        def backtrack(row):
            if row == n:
                res.append([''.join(r) for r in board])
                return
            for c in range(n):
                if cols[c] or diag1[row + c] or diag2[row - c + n]:
                    continue  # 冲突
                board[row][c] = 'Q'
                cols[c] = diag1[row + c] = diag2[row - c + n] = True
                backtrack(row + 1)
                board[row][c] = '.'
                cols[c] = diag1[row + c] = diag2[row - c + n] = False

        backtrack(0)
        return res
@@SOLUTION: 方法二：集合位运算标记
@@IDEA
位运算版：用三个整数按位表示列/对角线占用，冲突检查与撤销都是位运算（cols>>c & 1 等），常数效率最高；传值天然完成撤销。
@@JAVA
class Solution {
    public List<List<String>> solveNQueens(int n) {
        List<List<String>> res = new ArrayList<>();
        int[] queens = new int[n]; // queens[row] = col
        backtrack(res, queens, 0, 0, 0, 0, n);
        return res;
    }
    private void backtrack(List<List<String>> res, int[] queens, int row,
                           int cols, int d1, int d2, int n) {
        if (row == n) {
            List<String> solution = new ArrayList<>();
            for (int r = 0; r < n; r++) {
                StringBuilder sb = new StringBuilder();
                for (int c = 0; c < n; c++) sb.append(queens[r] == c ? 'Q' : '.');
                solution.add(sb.toString());
            }
            res.add(solution);
            return;
        }
        for (int c = 0; c < n; c++) {
            if (((cols >> c) & 1) == 1) continue;                      // 列冲突
            if (((d1 >> (row + c)) & 1) == 1) continue;                // 主对角线
            if (((d2 >> (row - c + n)) & 1) == 1) continue;            // 副对角线
            queens[row] = c; // 做选择
            backtrack(res, queens, row + 1,
                    cols | (1 << c), d1 | (1 << (row + c)), d2 | (1 << (row - c + n)), n);
            // 整数传值天然完成撤销
        }
    }
}

                board = []
                for r in range(n):
                    board.append('.' * queens[r] + 'Q' + '.' * (n - queens[r] - 1))
                res.append(board)
                return
            for c in range(n):
                if (cols >> c) & 1:              # 列冲突
                    continue
                if (d1 >> (row + c)) & 1:        # 主对角线
                    continue
                if (d2 >> (row - c + n)) & 1:    # 副对角线
                    continue
                backtrack(row + 1, cols | (1 << c),
                          d1 | (1 << (row + c)), d2 | (1 << (row - c + n)),
                          queens + [c])  # 传值天然撤销

        backtrack(0, 0, 0, 0, [])
        return res
@@SOLUTION: 方法三：逐行回溯 + 每行线性检查（无标记数组）
@@IDEA
朴素对照：不维护标记数组，放置前线性扫描之前所有行的皇后位置判断同列/对角线冲突。每步 O(n)，代码最短，帮助理解冲突定义。
@@JAVA
class Solution {
    public List<List<String>> solveNQueens(int n) {
        List<List<String>> res = new ArrayList<>();
        int[] queens = new int[n]; // queens[row] = col
        backtrack(res, queens, 0, n);
        return res;
    }
    private void backtrack(List<List<String>> res, int[] queens, int row, int n) {
        if (row == n) {
            List<String> solution = new ArrayList<>();
            for (int r = 0; r < n; r++) {
                StringBuilder sb = new StringBuilder();
                for (int c = 0; c < n; c++) sb.append(queens[r] == c ? 'Q' : '.');
                solution.add(sb.toString());
            }
            res.add(solution);
            return;
        }
        for (int c = 0; c < n; c++) {
            if (conflict(queens, row, c)) continue; // 与之前行冲突
            queens[row] = c;
            backtrack(res, queens, row + 1, n);
        }
    }
    private boolean conflict(int[] queens, int row, int col) {
        for (int prev = 0; prev < row; prev++) {
            if (queens[prev] == col) return true;              // 同列
            if (Math.abs(queens[prev] - col) == row - prev) return true; // 对角线
        }
        return false;
    }
}

class Solution:
    def solveNQueens(self, n: int) -> List[List[str]]:
        res = []

        def conflict(queens, row, col):  # 与之前行冲突
            for prev in range(row):
                if queens[prev] == col:                        # 同列
                    return True
                if abs(queens[prev] - col) == row - prev:      # 对角线
                    return True
            return False

        def backtrack(row, queens):
            if row == n:
                board = ['.' * q + 'Q' + '.' * (n - q - 1) for q in queens]
                res.append(board)
                return
            for c in range(n):
                if conflict(queens, row, c):
                    continue
                backtrack(row + 1, queens + [c])

        backtrack(0, [])
        return res
