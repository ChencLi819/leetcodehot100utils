给你两个单链表的头节点 headA 和 headB，找出并返回两链表相交的起始节点；不相交则返回 null。
整个链式结构中不存在环，函数返回结果后链表必须保持原样。
示例：两链表在节点 c1 相交（a 段长 5、b 段长 6）→ 返回 c1。
@@SOLUTION: 方法一：浪漫双指针（推荐，O(1) 空间）
@@IDEA
浪漫双指针：a 走完 A 转 B，b 走完 B 转 A；两者总路程相同（a+c = b+c），必然同步到达交点或同时为 null。O(n+m)/O(1)，无需测长度。
@@JAVA
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        ListNode a = headA, b = headB;
        while (a != b) {
            a = (a == null) ? headB : a.next; // a 走完 A 转 B
            b = (b == null) ? headA : b.next; // b 走完 B 转 A
        }
        return a; // 相遇点即交点，或同时为 null
    }
}

public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        Set<ListNode> seen = new HashSet<>();
        for (ListNode p = headA; p != null; p = p.next) {
            seen.add(p); // 记录 A 的所有节点
        }
        for (ListNode p = headB; p != null; p = p.next) {
            if (seen.contains(p)) return p; // 第个公共节点
        }
        return null;
    }
}

@@SOLUTION: 方法三：先对齐长度再同步前进
@@IDEA
先分别测出两链表长度，长链先走差值步对齐，再同步前进，第一个相同节点即交点。O(n+m)/O(1)，思路朴素好讲。
@@JAVA
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
        int la = length(headA), lb = length(headB);
        ListNode longL = la >= lb ? headA : headB;  // 长链表
        ListNode shortL = la >= lb ? headB : headA;
        for (int i = 0; i < Math.abs(la - lb); i++) {
            longL = longL.next; // 长链表先走差值步
        }
        while (longL != shortL) { // 同步前进
            longL = longL.next;
            shortL = shortL.next;
        }
        return longL;
    }
    private int length(ListNode head) {
        int n = 0;
        for (ListNode p = head; p != null; p = p.next) n++;
        return n;
    }
}

@@SOLUTION: 方法一：迭代三指针（推荐）
@@IDEA
迭代三指针：每步把 cur.next 指向 prev，然后 prev、cur 整体右移；先用临时变量暂存 next 防断链。O(n)/O(1)，链表基本功。

class Solution {
    public ListNode reverseList(ListNode head) {
        ListNode prev = null, cur = head;
        while (cur != null) {
            ListNode next = cur.next; // 暂存后继
            cur.next = prev;          // 反转当前指针
            prev = cur;
            cur = next;
        }
        return prev;
    }
}

class Solution {
    public ListNode reverseList(ListNode head) {
        if (head == null || head.next == null) return head; // 递归出口
        ListNode newHead = reverseList(head.next); // 反转后续
        head.next.next = head; // 后继回指自己（自己成新尾）
        head.next = null;      // 断开旧指向
        return newHead;
    }
}

            ListNode next = cur.next;
            cur.next = dummy.next; // 头插

            cur = next;

@@SOLUTION: 方法一：找中点 + 反转后半段（推荐，O(1) 空间）
@@IDEA
O(1) 空间标准解：快慢指针找中点，反转后半段，再与前半段逐节点比较。注意偶数长度时前后半差一个节点不影响判断。O(n)/O(1)。
@@JAVA
class Solution {
    public boolean isPalindrome(ListNode head) {
        ListNode slow = head, fast = head;
        while (fast != null && fast.next != null) { // 找中点
            slow = slow.next;
            fast = fast.next.next;
        }
        ListNode p1 = head, p2 = reverse(slow); // 反转后半段
        while (p2 != null) {                    // 逐节点比较
            if (p1.val != p2.val) return false;
            p1 = p1.next;
            p2 = p2.next;
        }
        return true;
    }
    private ListNode reverse(ListNode head) {
        ListNode prev = null, cur = head;
        while (cur != null) {
            ListNode next = cur.next;
            cur.next = prev;
            prev = cur;
            cur = next;
        }
        return prev;
    }
}

        slow = fast = head
        while fast and fast.next:   # 找中点
            slow = slow.next
            fast = fast.next.next
        p1, p2 = head, self._reverse(slow)  # 反转后半段
        while p2:                   # 逐节点比较
            if p1.val != p2.val:
                return False
            p1, p2 = p1.next, p2.next
        return True

    def _reverse(self, head):
        prev, cur = None, head
        while cur:
            cur.next, prev, cur = prev, cur, cur.next
        return prev

class Solution {
    public boolean isPalindrome(ListNode head) {
        List<Integer> vals = new ArrayList<>();
        for (ListNode p = head; p != null; p = p.next) vals.add(p.val);
        int l = 0, r = vals.size() - 1;
        while (l < r) { // 首尾双指针
            if (!vals.get(l).equals(vals.get(r))) return false;
            l++;
            r--;
        }
        return true;
    }
}

    public boolean isPalindrome(ListNode head) {
        front = head;
        return check(head);
    }
    private boolean check(ListNode node) {
        if (node != null) {
            if (!check(node.next)) return false;   // 先递归到尾
            if (node.val != front.val) return false; // 回溯时比较
            front = front.next;                     // 正序指针右移
        }
        return true;
    }
}

            if (slow == fast) return true; // 相遇即有环
        }
        return false;
    }

@@IDEA

public class Solution {
    public ListNode detectCycle(ListNode head) {
        ListNode slow = head, fast = head;
        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;
            if (slow == fast) {        // 相遇
                ListNode p = head;
                while (p != slow) {    // 同速前进找入环点
                    p = p.next;
                    slow = slow.next;
                }
                return p;
            }
        }
        return null;
    }
}

    public ListNode detectCycle(ListNode head) {
        // 1. 判环并得到环内一个节点
        ListNode slow = head, fast = head, inCycle = null;
        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;
            if (slow == fast) { inCycle = slow; break; }
        }
        if (inCycle == null) return null;
        // 2. 测环长
        int c = 1;
        for (ListNode p = inCycle.next; p != inCycle; p = p.next) c++;
        // 3. 一个指针从 head 先走 c 步，再与 head 同步——相遇即入环点
        ListNode ahead = head;
        while (c-- > 0) ahead = ahead.next;
        ListNode p = head;
        while (p != ahead) {
            p = p.next;
            ahead = ahead.next;
        }
        return p;
    }

    public ListNode reverseKGroup(ListNode head, int k) {
        List<ListNode> nodes = new ArrayList<>();
        for (ListNode p = head; p != null; p = p.next) nodes.add(p);
        int n = nodes.size();
        for (int start = 0; start + k <= n; start += k) {
            // 组内倒序重排
            for (int i = start, j = start + k - 1; i < j; i++, j--) {
                ListNode t = nodes.get(i); nodes.set(i, nodes.get(j)); nodes.set(j, t);
            }
        }
        for (int i = 0; i + 1 < n; i++) nodes.get(i).next = nodes.get(i + 1);
        if (!nodes.isEmpty()) nodes.get(n - 1).next = null;
        return nodes.isEmpty() ? null : nodes.get(0);
    }

class Solution {
    public ListNode sortList(ListNode head) {
        if (head == null) return null;
        int len = 0;
        for (ListNode p = head; p != null; p = p.next) len++;
        ListNode dummy = new ListNode(0, head);
        for (int step = 1; step < len; step <<= 1) {   // 子链长度翻倍
            ListNode prev = dummy, cur = dummy.next;
            while (cur != null) {
                ListNode left = cur;
                ListNode mid = cut(left, step);        // 第一段
                ListNode right = cut(mid, step);       // 第二段
                prev.next = merge(left, right);        // 合并两段
                while (prev.next != null) prev = prev.next; // 移到尾部
                cur = right;
            }
        }
        return dummy.next;
    }
    private ListNode cut(ListNode head, int n) { // 截断 n 个节点，返回剩余头

            head = head.next;
            n--;
        }
        if (head == null) return null;
        ListNode next = head.next;
        head.next = null;
        return next;
    }

            else { cur.next = b; b = b.next; }

    public ListNode mergeKLists(ListNode[] lists) {
        if (lists == null || lists.length == 0) return null;
        PriorityQueue<ListNode> heap = new PriorityQueue<>((a, b) -> a.val - b.val);
        for (ListNode h : lists) {
            if (h != null) heap.offer(h);

        ListNode dummy = new ListNode(0), cur = dummy;
        while (!heap.isEmpty()) {
            ListNode node = heap.poll(); // 当前最小
            cur.next = node;
            cur = cur.next;
            if (node.next != null) heap.offer(node.next); // 后继入堆

        return dummy.next;

                nxt.append(self._merge_two(a, b))  # 两两合并
            cur = nxt
        return cur[0]

    def _merge_two(self, a, b):
        dummy = tail = ListNode(0)
        while a and b:
            if a.val <= b.val:
                tail.next, a = a, a.next
            else:
                tail.next, b = b, b.next
            tail = tail.next
        tail.next = a or b
        return dummy.next

@@IDEA
全部节点收集到列表按值排序后顺序回接。O(N log N)/O(N)，代码最短，数据量小的时候最省事。

class Solution {
    public ListNode mergeKLists(ListNode[] lists) {
        List<ListNode> nodes = new ArrayList<>();
        for (ListNode h : lists) {
            for (ListNode p = h; p != null; p = p.next) nodes.add(p);
        }
        nodes.sort((a, b) -> a.val - b.val); // 按值排序
        ListNode dummy = new ListNode(0), cur = dummy;
        for (ListNode node : nodes) { // 回接
            cur.next = node;
            cur = cur.next;
        }
        cur.next = null;
        return dummy.next;
    }
}

class LRUCache {
    private final int cap;
    private final LinkedHashMap<Integer, Integer> map;

    public LRUCache(int capacity) {
        this.cap = capacity;
        this.map = new LinkedHashMap<>(16, 0.75f, true); // 按访问顺序
    }

    public int get(int key) {
        return map.getOrDefault(key, -1); // get 即视为访问
    }

    public void put(int key, int value) {
        map.put(key, value);
        if (map.size() > cap) {
            Integer eldest = map.keySet().iterator().next(); // 最久未使用
            map.remove(eldest);
        }
    }

@@IDEA

class LRUCache {
    private static class Node {
        int key, val;
        Node prev, next;
        Node(int k, int v) { key = k; val = v; }
    }
    private final int cap;
    private final Map<Integer, Node> map = new HashMap<>();
    private final Node head = new Node(0, 0), tail = new Node(0, 0); // 哨兵

    public LRUCache(int capacity) {
        this.cap = capacity;
        head.next = tail;
        tail.prev = head;
    }

    public int get(int key) {
        Node node = map.get(key);
        if (node == null) return -1;
        moveToFront(node); // 提升为最新
        return node.val;
    }

    public void put(int key, int value) {
        Node node = map.get(key);
        if (node != null) {
            node.val = value;
            moveToFront(node);
            return;
        }
        if (map.size() >= cap) {
            Node eldest = tail.prev;  // 最久未使用
            remove(eldest);
            map.remove(eldest.key);
        }
        node = new Node(key, value);
        map.put(key, node);
        addToFront(node);
    }

    private void remove(Node node) {
        node.prev.next = node.next;
        node.next.prev = node.prev;
    }
    private void addToFront(Node node) {
        node.next = head.next;
        node.prev = head;
        head.next.prev = node;
        head.next = node;
    }
    private void moveToFront(Node node) {
        remove(node);
        addToFront(node);
    }
}

class LRUCache:
    class Node:
        __slots__ = ("key", "val", "prev", "next")

        def __init__(self, k=0, v=0):
            self.key, self.val = k, v
            self.prev = self.next = None

    def __init__(self, capacity: int):
        self.cap = capacity
        self.map = {}
        self.head = self.Node()  # 哨兵
        self.tail = self.Node()
        self.head.next = self.tail
        self.tail.prev = self.head

    def get(self, key: int) -> int:
        if key not in self.map:
            return -1
        node = self.map[key]
        self._remove(node)
        self._add_front(node)  # 提升为最新
        return node.val

    def put(self, key: int, value: int) -> None:
        if key in self.map:
            node = self.map[key]
            node.val = value
            self._remove(node)
            self._add_front(node)
            return
        if len(self.map) >= self.cap:
            eldest = self.tail.prev  # 最久未使用
            self._remove(eldest)
            del self.map[eldest.key]
        node = self.Node(key, value)
        self.map[key] = node
        self._add_front(node)

    def _remove(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_front(self, node):
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
@@SOLUTION: 方法三：单链表 + 哈希记前驱（O(n) 最坏，作对照）
@@IDEA
简化对照：单链表/列表按访问时间排序 + 哈希查值，命中后移动到尾部。定位要遍历，最坏 O(n)，仅用于理解「顺序维护」的本质。
@@JAVA
class LRUCache {
    private static class Node {
        int key, val;
        Node next;
        Node(int k, int v) { key = k; val = v; }
    }
    private final int cap;
    private final Node dummy = new Node(0, 0); // 哨兵头，头部=最久未使用
    private Node tail = dummy;
    private int size = 0;
    private final Map<Integer, Node> prevMap = new HashMap<>(); // key -> 前驱节点

    public LRUCache(int capacity) {
        this.cap = capacity;
    }

    public int get(int key) {
        if (!prevMap.containsKey(key)) return -1;
        promote(key);   // 移到尾部=最新
        return tail.val;
    }

    public void put(int key, int value) {
        if (prevMap.containsKey(key)) {
            promote(key);
            tail.val = value;
            return;
        }
        if (size == cap) {
            Node eldest = dummy.next;              // 淘汰头部
            prevMap.remove(eldest.key);
            if (eldest.next != null) {
                prevMap.put(eldest.next.key, dummy); // 新头的前驱=哨兵
            }
            dummy.next = eldest.next;
            size--;
        }
        Node node = new Node(key, value);
        tail.next = node;
        prevMap.put(key, tail);
        tail = node;
        size++;
    }

    private void promote(int key) { // 摘下节点接到尾部
        Node prev = prevMap.get(key);
        Node node = prev.next;
        if (node == tail) return;
        prev.next = node.next;
        prevMap.put(node.next.key, prev); // 更新后继的前驱
        tail.next = node;
        prevMap.put(key, tail);
        tail = node;
        tail.next = null;
    }
}

class LRUCache:
    """单链表按访问时间排序 + 哈希记前驱（概念对照版，简化为列表实现）。

    访问时需遍历定位节点（最坏 O(n)），非严格 O(1)；
    用于理解 LRU 的顺序维护本质，生产实现见方法一/二。
    """

    def __init__(self, capacity: int):
        self.cap = capacity
        self.order = []  # key 列表，尾部=最近使用
        self.store = {}

    def get(self, key: int) -> int:
        if key not in self.store:
            return -1
        self.order.remove(key)  # 摘除
        self.order.append(key)  # 标记为最近使用
        return self.store[key]

    def put(self, key: int, value: int) -> None:
        if key in self.order:
            self.order.remove(key)
        self.order.append(key)
        self.store[key] = value
        if len(self.order) > self.cap:  # 淘汰最久未使用
            eldest = self.order.pop(0)
            del self.store[eldest]
