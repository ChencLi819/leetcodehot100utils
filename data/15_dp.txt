class Solution {
    public int climbStairs(int n) {
        int a = 1, b = 1; // f(0)=1, f(1)=1
        for (int i = 2; i <= n; i++) {
            int c = a + b;
            a = b;
            b = c;
        }
        return b;
    }
}

class Solution {
    public int climbStairs(int n) {
        int[][] base = {{1, 1}, {1, 0}};
        int[][] res = pow(base, n); // res = base^n
        // f(n) = res[0][0] * f(1) + res[0][1] * f(0)，f(0)=f(1)=1
        return res[0][0] + res[0][1];
    }
    private int[][] pow(int[][] m, int p) {
        int[][] result = {{1, 0}, {0, 1}}; // 单位阵
        while (p > 0) {
            if ((p & 1) == 1) result = mul(result, m);
            m = mul(m, m); // 平方
            p >>= 1;
        }
        return result;
    }
    private int[][] mul(int[][] a, int[][] b) {
        int[][] c = new int[2][2];
        for (int i = 0; i < 2; i++) {
            for (int k = 0; k < 2; k++) {
                if (a[i][k] == 0) continue;
                for (int j = 0; j < 2; j++) {
                    c[i][j] += a[i][k] * b[k][j];
                }
            }
        }
        return c;
    }
}

class Solution {
    public List<List<Integer>> generate(int numRows) {
        List<List<Integer>> res = new ArrayList<>();
        for (int i = 0; i < numRows; i++) {
            List<Integer> row = new ArrayList<>();
            for (int j = 0; j <= i; j++) {
                if (j == 0 || j == i) row.add(1); // 两端为 1
                else row.add(res.get(i - 1).get(j - 1) + res.get(i - 1).get(j));
            }
            res.add(row);
        }
        return res;
    }
}

            long c = 1; // C(i, 0) = 1

class Solution {
    public List<List<Integer>> generate(int numRows) {
        List<List<Integer>> res = new ArrayList<>();
        int[] row = new int[numRows];
        for (int i = 0; i < numRows; i++) {
            for (int j = i; j >= 1; j--) {  // 从后往前更新
                row[j] += row[j - 1];
            }
            row[i] = 1; // 行末置 1
            List<Integer> line = new ArrayList<>();
            for (int j = 0; j <= i; j++) line.add(row[j]);
            res.add(line);
        }
        return res;
    }

@@IDEA

class Solution {
    public int rob(int[] nums) {
        int prev2 = 0, prev1 = 0; // dp[i-2], dp[i-1]
        for (int x : nums) {
            int cur = Math.max(prev1, prev2 + x);
            prev2 = prev1;
            prev1 = cur;
        }
        return prev1;
    }
}

class Solution {
    public int rob(int[] nums) {
        int robEven = 0, robOdd = 0;
        for (int i = 0; i < nums.length; i++) {
            if (i % 2 == 0) {
                robEven = Math.max(robEven + nums[i], robOdd); // 偶位累计 vs 奇位最优
            } else {
                robOdd = Math.max(robOdd + nums[i], robEven);
            }
        }
        return Math.max(robEven, robOdd);
    }
}

@@IDEA

class Solution {
    public int numSquares(int n) {
        int[] dp = new int[n + 1];
        Arrays.fill(dp, Integer.MAX_VALUE);
        dp[0] = 0;
        for (int i = 1; i * i <= n; i++) {     // 物品：i*i
            for (int j = i * i; j <= n; j++) { // 容量：n
                dp[j] = Math.min(dp[j], dp[j - i * i] + 1);
            }
        }
        return dp[n];
    }
}

@@IDEA
BFS 最短路：0..n 为节点，x → x+i² 连边权 1，BFS 求从 0 到 n 的最短步数即最少数量；visited 防重复。O(n·√n)。

class Solution {
    public int numSquares(int n) {
        Deque<Integer> queue = new ArrayDeque<>();
        boolean[] visited = new boolean[n + 1];
        queue.offer(0);
        visited[0] = true;
        int steps = 0;
        while (!queue.isEmpty()) {
            steps++;
            int size = queue.size();
            for (int k = 0; k < size; k++) {
                int cur = queue.poll();
                for (int i = 1; cur + i * i <= n; i++) {
                    int next = cur + i * i;
                    if (next == n) return steps;   // 首次到达
                    if (!visited[next]) {
                        visited[next] = true;
                        queue.offer(next);
                    }
                }
            }
        }
        return n; // n 本身由 n 个 1 组成
    }
}

class Solution {
    public int numSquares(int n) {
        if (isSquare(n)) return 1;                       // 1 个
        int m = n;
        while (m % 4 == 0) m /= 4;                       // 剥离 4 的因子
        if (m % 8 == 7) return 4;                        // 4k+7 型 → 4 个
        for (int a = 1; a * a <= n; a++) {
            if (isSquare(n - a * a)) return 2;           // 可拆两个 → 2 个
        }
        return 3;                                        // 其余 → 3 个
    }
    private boolean isSquare(int x) {
        int r = (int) Math.sqrt(x);
        return r * r == x;
    }
}

class Solution {
    public int coinChange(int[] coins, int amount) {
        int[] dp = new int[amount + 1];
        Arrays.fill(dp, amount + 1); // 超过任何合法解的哨兵
        dp[0] = 0;
        for (int coin : coins) {
            for (int j = coin; j <= amount; j++) { // 完全背包正序
                dp[j] = Math.min(dp[j], dp[j - coin] + 1);
            }
        }
        return dp[amount] > amount ? -1 : dp[amount];
    }
}

    public int coinChange(int[] coins, int amount) {

        Deque<Integer> queue = new ArrayDeque<>();
        boolean[] visited = new boolean[amount + 1];
        queue.offer(0);
        visited[0] = true;
        int steps = 0;

            steps++;
            int size = queue.size();
            for (int k = 0; k < size; k++) {
                int cur = queue.poll();
                for (int coin : coins) {
                    int next = cur + coin;

                    if (next < amount && !visited[next]) {
                        visited[next] = true;
                        queue.offer(next);
                    }
                }
            }

class Solution {
    public int lengthOfLIS(int[] nums) {
        List<Integer> tails = new ArrayList<>(); // 各长度最小结尾
        for (int x : nums) {
            int lo = 0, hi = tails.size();       // 找第一个 >= x 的位置
            while (lo < hi) {
                int mid = (lo + hi) >>> 1;
                if (tails.get(mid) < x) lo = mid + 1;
                else hi = mid;
            }
            if (lo == tails.size()) tails.add(x); // 追加：LIS 变长
            else tails.set(lo, x);                // 替换：让结尾更小
        }
        return tails.size();

class Solution {
    public int lengthOfLIS(int[] nums) {
        // 离散化到 1..m
        int[] sorted = nums.clone();
        Arrays.sort(sorted);
        Map<Integer, Integer> rank = new HashMap<>();
        int m = 0;
        for (int v : sorted) {
            rank.putIfAbsent(v, ++m);
        }
        int[] tree = new int[m + 1]; // 树状数组维护前缀最大 dp
        int best = 0;
        for (int x : nums) {
            int r = rank.get(x);
            int cur = query(tree, r - 1) + 1; // 值 < x 的最大 dp + 1
            update(tree, r, cur);
            best = Math.max(best, cur);
        }
        return best;
    }
    private int query(int[] tree, int i) { // 前缀最大 [1..i]
        int res = 0;
        for (; i > 0; i -= i & (-i)) res = Math.max(res, tree[i]);
        return res;
    }
    private void update(int[] tree, int i, int val) { // 单点更新取 max
        for (; i < tree.length; i += i & (-i)) tree[i] = Math.max(tree[i], val);
    }

class Solution {
    public boolean canPartition(int[] nums) {
        int sum = 0;
        for (int x : nums) sum += x;
        if (sum % 2 != 0) return false;   // 奇数无解
        int target = sum / 2;
        boolean[] dp = new boolean[target + 1];
        dp[0] = true;
        for (int x : nums) {
            for (int j = target; j >= x; j--) { // 倒序：每数只用一次
                dp[j] = dp[j] || dp[j - x];
            }
        }
        return dp[target];
    }
}

@@IDEA
记忆化搜索：dfs(i, remain) = 用 nums[i..] 能否凑出 remain，选/不选两种转移，缓存状态。O(n×sum) 状态数。

class Solution {
    private int[] nums;
    private Boolean[][] memo;

    public boolean canPartition(int[] nums) {
        int sum = 0;
        for (int x : nums) sum += x;
        if (sum % 2 != 0) return false;
        this.nums = nums;
        this.memo = new Boolean[nums.length][sum / 2 + 1];
        return dfs(0, sum / 2);
    }
    private boolean dfs(int i, int remain) {
        if (remain == 0) return true;
        if (i >= nums.length || remain < 0) return false;
        if (memo[i][remain] != null) return memo[i][remain];
        // 选 / 不选 nums[i]
        return memo[i][remain] = dfs(i + 1, remain - nums[i]) || dfs(i + 1, remain);
    }
}

@@SOLUTION: 方法一：栈底哨兵法（推荐）
@@IDEA
栈底哨兵：栈存下标，先压 -1 作「最近未匹配位置」；遇 ')' 弹栈——弹空则当前下标成为新基底，否则长度 = i - 新栈顶。O(n)/O(n)。
@@JAVA
class Solution {
    public int longestValidParentheses(String s) {
        Deque<Integer> stack = new ArrayDeque<>();
        stack.push(-1); // 哨兵：最近未匹配位置
        int best = 0;
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == '(') {
                stack.push(i);
            } else {
                stack.pop();
                if (stack.isEmpty()) {
                    stack.push(i); // 新的未匹配基底
                } else {
                    best = Math.max(best, i - stack.peek());
                }
            }
        }
        return best;
    }
}

@@SOLUTION: 方法二：动态规划
@@IDEA
区间 DP：dp[i] = 以 i 结尾的最长有效长度。s[i]==')' 且 s[i-1]=='(' 时 +2；否则看 i-1-dp[i-1] 是否为 '('，是则连上更早的 dp。O(n)/O(n)。
@@JAVA
class Solution {
    public int longestValidParentheses(String s) {
        int n = s.length(), best = 0;
        int[] dp = new int[n]; // dp[i]: 以 i 结尾的最长有效长度
        for (int i = 1; i < n; i++) {
            if (s.charAt(i) == ')') {
                if (s.charAt(i - 1) == '(') {
                    dp[i] = (i >= 2 ? dp[i - 2] : 0) + 2; // "…()"
                } else if (i - dp[i - 1] - 1 >= 0
                        && s.charAt(i - dp[i - 1] - 1) == '(') {
                    // "((…))"：跳过内层有效段再连
                    int head = i - dp[i - 1] - 1;
                    dp[i] = dp[i - 1] + 2 + (head > 0 ? dp[head - 1] : 0);
                }
                best = Math.max(best, dp[i]);
            }
        }
        return best;
    }
}

@@SOLUTION: 方法三：双向扫描计数
@@IDEA
双向扫描计数：左→右扫 cnt( 与 cnt)，平衡时结算、右多则重置；再右→左对称扫一遍（解决 "(()" 类漏判）。O(n)/O(1)，无栈无 DP。
@@JAVA
class Solution {
    public int longestValidParentheses(String s) {
        int best = 0;
        int l = 0, r = 0;
        for (int i = 0; i < s.length(); i++) { // 左→右
            if (s.charAt(i) == '(') l++;
            else r++;
            if (l == r) best = Math.max(best, 2 * r); // 平衡
            else if (r > l) l = r = 0;                // 右括号过多，重置
        }
        l = r = 0;
        for (int i = s.length() - 1; i >= 0; i--) { // 右→左
            if (s.charAt(i) == '(') l++;
            else r++;
            if (l == r) best = Math.max(best, 2 * l);
            else if (l > r) l = r = 0;               // 左括号过多，重置
        }
        return best;
    }
}
